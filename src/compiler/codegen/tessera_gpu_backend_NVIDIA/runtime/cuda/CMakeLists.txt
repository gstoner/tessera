# Shipped NVIDIA mma.sync GEMM runtime symbols (tessera_nvidia_mma_gemm_*).
# The NVIDIA analog of Tessera_ROCM_Backend/runtime/hip. Built only when CUDA is
# available: f16 has a versioned SM120 cubin peer and every other dtype
# NVRTC-compiles for the live device arch at first use. f16 retains the same
# canonical source as an NVRTC fallback, so a deployment never needs to trust a
# stale binary. This is the shipped symbol the backend_manifest
# `hardware_verified` contract requires; the execute-compare fixture dlopens and
# numerically validates each dtype against a numpy reference.
if(TESSERA_ENABLE_CUDA)
  if(NOT TARGET CUDA::cudart)
    find_package(CUDAToolkit QUIET)
  endif()
  # NVIDIA-AOT-1: f16 is the first production AOT peer.  The generated header
  # is deliberately made from the exact .cu file passed to nvcc, so the cubin
  # and NVRTC fallback cannot silently diverge in source or argument layout.
  set(_tessera_aot_dir "${CMAKE_CURRENT_SOURCE_DIR}/aot")
  set(_tessera_aot_f16_source
      "${_tessera_aot_dir}/tessera_nvidia_mma_f16_sm120_v1.cu")
  set(_tessera_aot_f16_cubin
      "${CMAKE_CURRENT_BINARY_DIR}/tessera_nvidia_mma_f16_sm120_v1.cubin")
  set(_tessera_aot_f16_fatbin
      "${CMAKE_CURRENT_BINARY_DIR}/tessera_nvidia_mma_f16_sm120_v1.fatbin")
  set(_tessera_aot_f16_header
      "${CMAKE_CURRENT_BINARY_DIR}/tessera_nvidia_mma_f16_sm120_v1_source.h")
  set(_tessera_aot_f16_manifest
      "${CMAKE_CURRENT_BINARY_DIR}/tessera_nvidia_mma_f16_sm120_v1.json")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
               "${_tessera_aot_f16_source}")
  file(READ "${_tessera_aot_f16_source}" _tessera_aot_f16_text)
  file(SHA256 "${_tessera_aot_f16_source}" _tessera_aot_f16_sha256)
  file(WRITE "${_tessera_aot_f16_header}"
"// Generated by CMake from tessera_nvidia_mma_f16_sm120_v1.cu.\n"
"inline constexpr char kSrcF16Aot[] = R\"TESSERA_AOT(\n${_tessera_aot_f16_text}\n)TESSERA_AOT\";\n"
"inline constexpr char kSm120F16AotSourceSha256[] = \"${_tessera_aot_f16_sha256}\";\n")
  file(WRITE "${_tessera_aot_f16_manifest}"
"{\n"
"  \"schema\": \"tessera.nvidia.aot.package.v1\",\n"
"  \"artifact_id\": \"tessera_nvidia_mma_f16_sm120\",\n"
"  \"artifact_version\": 1,\n"
"  \"abi_version\": 1,\n"
"  \"formats\": [\"fatbin\", \"cubin\"],\n"
"  \"target\": \"nvidia_sm120\",\n"
"  \"architecture\": \"sm_120a\",\n"
"  \"compute_capability\": [12, 0],\n"
"  \"minimum_cuda_driver\": 13000,\n"
"  \"entry\": \"gemm\",\n"
"  \"source_sha256\": \"${_tessera_aot_f16_sha256}\",\n"
"  \"fallback\": \"NVRTC compiles the identical canonical .cu source\"\n"
"}\n")
  set(_tessera_aot_definitions
      "-DTESSERA_AOT_ARTIFACT_VERSION=1"
      "-DTESSERA_AOT_ABI_VERSION=1"
      "-DTESSERA_AOT_SOURCE_SHA256=\\\"${_tessera_aot_f16_sha256}\\\"")
  add_custom_command(
    OUTPUT "${_tessera_aot_f16_cubin}"
    COMMAND "${CMAKE_CUDA_COMPILER}" -std=c++17 -O3 -arch=sm_120a -cubin
            ${_tessera_aot_definitions}
            "${_tessera_aot_f16_source}" -o "${_tessera_aot_f16_cubin}"
    DEPENDS "${_tessera_aot_f16_source}"
    VERBATIM
    COMMENT "Building versioned SM120 f16 AOT cubin")
  add_custom_command(
    OUTPUT "${_tessera_aot_f16_fatbin}"
    COMMAND "${CMAKE_CUDA_COMPILER}" -std=c++17 -O3 -arch=sm_120a -fatbin
            ${_tessera_aot_definitions}
            "${_tessera_aot_f16_source}" -o "${_tessera_aot_f16_fatbin}"
    DEPENDS "${_tessera_aot_f16_source}"
    VERBATIM
    COMMENT "Building versioned SM120 f16 AOT fatbin")
  add_custom_target(tessera_nvidia_sm120_f16_aot_artifact
                    DEPENDS "${_tessera_aot_f16_cubin}"
                            "${_tessera_aot_f16_fatbin}"
                            "${_tessera_aot_f16_manifest}")

  add_library(tessera_nvidia_gemm SHARED tessera_nvidia_gemm.cpp)
  add_dependencies(tessera_nvidia_gemm tessera_nvidia_sm120_f16_aot_artifact)
  target_include_directories(tessera_nvidia_gemm PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  add_custom_command(TARGET tessera_nvidia_gemm POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${_tessera_aot_f16_cubin}"
            "$<TARGET_FILE_DIR:tessera_nvidia_gemm>/tessera_nvidia_mma_f16_sm120_v1.cubin"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${_tessera_aot_f16_fatbin}"
            "$<TARGET_FILE_DIR:tessera_nvidia_gemm>/tessera_nvidia_mma_f16_sm120_v1.fatbin"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${_tessera_aot_f16_manifest}"
            "$<TARGET_FILE_DIR:tessera_nvidia_gemm>/tessera_nvidia_mma_f16_sm120_v1.json"
    VERBATIM)

  # Driver API (libcuda) — prefer the toolkit imported target, else the stub.
  if(TARGET CUDA::cuda_driver)
    target_link_libraries(tessera_nvidia_gemm PUBLIC CUDA::cuda_driver)
  else()
    target_link_libraries(tessera_nvidia_gemm PUBLIC cuda)
  endif()

  # NVRTC — imported target, else find the lib by name.
  if(TARGET CUDA::nvrtc)
    target_link_libraries(tessera_nvidia_gemm PUBLIC CUDA::nvrtc)
  else()
    find_library(TESSERA_NVRTC_LIB NAMES nvrtc
                 HINTS ${CUDAToolkit_LIBRARY_DIR} /usr/local/cuda/lib64)
    if(TESSERA_NVRTC_LIB)
      target_link_libraries(tessera_nvidia_gemm PUBLIC ${TESSERA_NVRTC_LIB})
    endif()
  endif()

  # cuda.h / nvrtc.h — imported target carries these; fall back to a CUDA prefix.
  if(CUDAToolkit_FOUND)
    target_include_directories(tessera_nvidia_gemm PUBLIC ${CUDAToolkit_INCLUDE_DIRS})
  else()
    if(DEFINED ENV{CUDA_PATH})
      set(_tessera_cuda_prefix "$ENV{CUDA_PATH}")
    else()
      set(_tessera_cuda_prefix "/usr/local/cuda")
    endif()
    target_include_directories(tessera_nvidia_gemm PUBLIC "${_tessera_cuda_prefix}/include")
    message(STATUS "tessera_nvidia_gemm: using CUDA include ${_tessera_cuda_prefix}/include (CUDAToolkit not found)")
  endif()

  # Stateless Philox/key-counter package. This is a separate ABI from GEMM so
  # RNG consumers do not acquire MMA/NVRTC state or workspace implicitly.
  add_library(tessera_nvidia_rng SHARED tessera_nvidia_rng.cu)
  target_link_libraries(tessera_nvidia_rng PUBLIC CUDA::cudart)
  set_target_properties(tessera_nvidia_rng PROPERTIES
    CUDA_ARCHITECTURES 120
    CUDA_STANDARD 17
    CUDA_STANDARD_REQUIRED ON)

  add_library(tessera_nvidia_fft SHARED
    tessera_nvidia_fft.cu
    tessera_nvidia_spectral.cu)
  target_link_libraries(tessera_nvidia_fft PUBLIC CUDA::cudart CUDA::cufft)
  set_target_properties(tessera_nvidia_fft PROPERTIES
    CUDA_ARCHITECTURES 120
    CUDA_STANDARD 17
    CUDA_STANDARD_REQUIRED ON)

  # C2-tail PTX launch bridge (COMPILER_REFACTOR_PLAN) — the emit-path launcher,
  # counterpart to apple_gpu_runtime.mm. Driver API only (JITs the emitted PTX;
  # no NVRTC device pass). References the core-runtime tsrRegisterGpuLauncher seam
  # (resolved by the hosting binary — the shipped .so keeps it undefined, exactly
  # like a plugin, so the direct register/invoke path dlopens standalone).
  add_library(tessera_nvidia_ptx_launch SHARED tessera_nvidia_ptx_launch.cpp)
  target_include_directories(tessera_nvidia_ptx_launch PUBLIC
    "${CMAKE_SOURCE_DIR}/src/runtime/include")
  if(TARGET CUDA::cuda_driver)
    target_link_libraries(tessera_nvidia_ptx_launch PUBLIC CUDA::cuda_driver)
  else()
    target_link_libraries(tessera_nvidia_ptx_launch PUBLIC cuda)
  endif()
  if(CUDAToolkit_FOUND)
    target_include_directories(tessera_nvidia_ptx_launch PUBLIC ${CUDAToolkit_INCLUDE_DIRS})
  else()
    target_include_directories(tessera_nvidia_ptx_launch PUBLIC "${_tessera_cuda_prefix}/include")
  endif()
endif()
